Create new file: `density_analysis.py`

```python
import pandas as pd
import numpy as np
from sklearn.cluster import DBSCAN

def calculate_density_score(df, commodity=None):
    """Calculate resource density score for each site"""
    
    # Filter by commodity if specified
    if commodity and commodity != 'All':
        df = df[df['Primary_Commodity'] == commodity].copy()
    
    # Score components (0-100 scale)
    scores = pd.DataFrame(index=df.index)
    
    # 1. Reserve presence (40 points)
    scores['reserve_score'] = df['Reserves'].notna().astype(int) * 40
    
    # 2. Activity status (30 points)
    status_map = {
        'Active': 30,
        'Active (Small Mining)': 25,
        'Feasibility Evaluation': 20,
        'Dormant': 10,
        'Unknown': 5,
        'Inactive': 0
    }
    scores['status_score'] = df['Status'].map(status_map).fillna(5)
    
    # 3. Multi-commodity bonus (15 points)
    scores['commodity_score'] = (
        (df['Commodity_2'].notna().astype(int) * 10) +
        (df['Commodity_3'].notna().astype(int) * 5)
    )
    
    # 4. Geological classification confidence (15 points)
    scores['geology_score'] = df['Geology_Classification'].notna().astype(int) * 15
    
    # Total density score
    df['Density_Score'] = scores.sum(axis=1)
    
    return df

def rank_sites_by_density(df, commodity='All', n=10, mode='top'):
    """Rank sites by density score"""
    
    df_scored = calculate_density_score(df, commodity)
    df_sorted = df_scored.sort_values('Density_Score', ascending=(mode == 'bottom'))
    
    return df_sorted.head(n)

def create_density_clusters(df, eps_km=20, min_samples=5):
    """Identify resource clusters using DBSCAN"""
    
    coords = df[['Latitude', 'Longitude']].values
    
    # DBSCAN clustering (eps in degrees, roughly 20km)
    clustering = DBSCAN(eps=eps_km/111, min_samples=min_samples).fit(coords)
    
    df['Cluster_ID'] = clustering.labels_
    
    # Calculate cluster statistics
    cluster_stats = []
    for cluster_id in df['Cluster_ID'].unique():
        if cluster_id == -1:  # Skip noise points
            continue
        
        cluster_data = df[df['Cluster_ID'] == cluster_id]
        
        stats = {
            'Cluster_ID': cluster_id,
            'Site_Count': len(cluster_data),
            'Center_Lat': cluster_data['Latitude'].mean(),
            'Center_Lon': cluster_data['Longitude'].mean(),
            'Dominant_Commodity': cluster_data['Primary_Commodity'].mode()[0],
            'Avg_Density_Score': cluster_data['Density_Score'].mean()
        }
        cluster_stats.append(stats)
    
    return df, pd.DataFrame(cluster_stats)
```

```python
import pandas as pd
import numpy as np
from scipy.spatial import ConvexHull
from shapely.geometry import LineString, Point
from sklearn.decomposition import PCA

def identify_mineral_belts(df, commodity, buffer_km=30):
    """Identify linear mineral belts for a commodity"""
    
    # Filter to specific commodity
    commodity_sites = df[df['Primary_Commodity'] == commodity].copy()
    
    if len(commodity_sites) < 3:
        return None
    
    # Get coordinates
    coords = commodity_sites[['Longitude', 'Latitude']].values
    
    # PCA to find principal direction
    pca = PCA(n_components=2)
    pca.fit(coords)
    
    # Principal direction (vein strike)
    direction = pca.components_[0]
    bearing = np.degrees(np.arctan2(direction[1], direction[0]))
    
    # Project points onto principal axis
    transformed = pca.transform(coords)
    
    return {
        'commodity': commodity,
        'bearing': bearing % 360,
        'length_km': (transformed[:, 0].max() - transformed[:, 0].min()) * 111,
        'width_km': (transformed[:, 1].max() - transformed[:, 1].min()) * 111,
        'site_count': len(commodity_sites),
        'center_lat': commodity_sites['Latitude'].mean(),
        'center_lon': commodity_sites['Longitude'].mean()
    }

def create_trend_lines(df, commodity):
    """Create trend line coordinates for visualization"""
    
    commodity_sites = df[df['Primary_Commodity'] == commodity]
    
    if len(commodity_sites) < 2:
        return None
    
    coords = commodity_sites[['Longitude', 'Latitude']].values
    
    # PCA for trend direction
    pca = PCA(n_components=1)
    pca.fit(coords)
    
    # Create line extending through the data
    mean_point = coords.mean(axis=0)
    direction = pca.components_[0]
    
    # Extend line to cover all points
    max_distance = np.max(np.abs(coords - mean_point))
    
    start_point = mean_point - direction * max_distance * 1.2
    end_point = mean_point + direction * max_distance * 1.2
    
    return {
        'start': start_point,
        'end': end_point,
        'commodity': commodity
    }

def analyze_geology_patterns(df):
    """Analyze geological formation patterns"""
    
    patterns = []
    
    # Group by geology type
    for geology_type in df['Geology_Classification'].dropna().unique():
        geology_sites = df[df['Geology_Classification'] == geology_type]
        
        if len(geology_sites) < 5:
            continue
        
        pattern = {
            'geology_type': geology_type,
            'site_count': len(geology_sites),
            'commodities': geology_sites['Primary_Commodity'].value_counts().to_dict(),
            'provinces': geology_sites['Province'].value_counts().to_dict(),
            'avg_lat': geology_sites['Latitude'].mean(),
            'avg_lon': geology_sites['Longitude'].mean(),
            'spatial_extent_km': calculate_spatial_extent(geology_sites)
        }
        
        patterns.append(pattern)
    
    return pd.DataFrame(patterns)

def calculate_spatial_extent(sites):
    """Calculate the geographic extent of sites"""
    lat_range = sites['Latitude'].max() - sites['Latitude'].min()
    lon_range = sites['Longitude'].max() - sites['Longitude'].min()
    
    # Convert to km (rough approximation)
    extent = np.sqrt((lat_range * 111)**2 + (lon_range * 111)**2)
    
    return extent
```

STEP 1: Create Accessibility Scoring (2-3 hours)

Create new file: `accessibility.py`

```python
import pandas as pd
import numpy as np

def calculate_accessibility_score(df):
    """Calculate accessibility score for each site"""
    
    scores = pd.DataFrame(index=df.index)
    
    # 1. Distance from Chingola (40 points, inverse)
    max_distance = df['Distance_From_Chingola_km'].max()
    scores['chingola_score'] = (
        40 * (1 - df['Distance_From_Chingola_km'] / max_distance)
    ).fillna(20)
    
    # 2. Travel time (30 points, inverse)
    max_time = df['Travel_Time_From_Chingola_Hours'].max()
    scores['travel_score'] = (
        30 * (1 - df['Travel_Time_From_Chingola_Hours'] / max_time)
    ).fillna(15)
    
    # 3. Distance to nearest center (20 points, inverse)
    if 'Distance_Nearest_Center_km' in df.columns:
        max_center_dist = df['Distance_Nearest_Center_km'].max()
        scores['center_score'] = (
            20 * (1 - df['Distance_Nearest_Center_km'] / max_center_dist)
        ).fillna(10)
    else:
        scores['center_score'] = 10
    
    # 4. Province infrastructure bonus (10 points)
    # Provinces with better infrastructure
    province_scores = {
        'Copperbelt Province': 10,
        'Lusaka Province': 9,
        'Central Province': 7,
        'Southern Province': 6,
        'Northern Province': 5,
        'Eastern Province': 5,
        'Muchinga Province': 4,
        'North-Western Province': 4,
        'Luapula Province': 3,
        'Western Province': 3
    }
    scores['province_score'] = df['Province'].map(province_scores).fillna(5)
    
    # Total accessibility score (0-100)
    df['Accessibility_Score'] = scores.sum(axis=1)
    
    # Categorize
    df['Access_Category'] = pd.cut(
        df['Accessibility_Score'],
        bins=[0, 40, 60, 80, 100],
        labels=['Poor', 'Fair', 'Good', 'Excellent']
    )
    
    return df

def rank_by_accessibility(df, n=20):
    """Get top N most accessible sites"""
    return df.nlargest(n, 'Accessibility_Score')
```
